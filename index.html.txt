    <script>
        let currentPrice = 129.45; let totalBalance = 0.00;
        setInterval(() => {
            let change = (Math.random() - 0.5) * 0.15;
            currentPrice = parseFloat((currentPrice + change).toFixed(2));
            document.getElementById('live-price').innerText = currentPrice + ' KES';
        }, 3000);
        async function triggerMpesaDeposit() {
            const phone = document.getElementById('phone-input').value;
            const amt = document.getElementById('amount-input').value;
            if(!phone || !amt) { alert('Please enter phone and amount.'); return; }
            try {
                const response = await fetch('/api/v1/payments/stk-push', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ phone_number: phone, amount: parseFloat(amt) })
                });
                const data = await response.json();
                alert(data.message);
                if(data.status === 'Success') {
                    totalBalance += parseFloat(amt);
                    document.getElementById('wallet-balance').innerText = 'KES ' + totalBalance.toLocaleString('en-US', {minimumFractionDigits: 2});
                    addLedgerRow(data.ref_id, 'MPESA_DEPOSIT', 'KES +' + parseFloat(amt).toLocaleString(), 'SETTLED', 'txt-green');
                }
            } catch (error) { alert('API Error.'); }
        }
        function executeTrade(type) {
            if (totalBalance <= 0) { alert('Please top up your wallet via M-Pesa first.'); return; }
            let tradeAmount = 500; let ref = 'TX_' + Math.random().toString(36).substring(2, 9).toUpperCase();
            let colorClass = type === 'BUY' ? 'txt-green' : 'txt-red';
            addLedgerRow(ref, type, 'KES ' + tradeAmount.toLocaleString(), 'RUNNING', colorClass);
        }
        function addLedgerRow(ref, action, amount, status, styleClass) {
            const body = document.getElementById('ledger-body'); const row = document.createElement('tr'); let time = new Date().toLocaleTimeString();
            row.innerHTML = `<td>${time}</td><td>${ref}</td><td class="${styleClass}">${action}</td><td>${amount}</td><td class="txt-green">${status}</td>`;
            body.insertBefore(row, body.firstChild);
        }
